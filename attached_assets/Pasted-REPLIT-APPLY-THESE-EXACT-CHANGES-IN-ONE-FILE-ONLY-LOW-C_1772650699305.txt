const getPointer = (ev: PointerEvent) => {
  const rect = renderer.domElement.getBoundingClientRect();
  pointer.x = ((ev.clientX - rect.left) / rect.width) * 2 - 1;
  pointer.y = -(((ev.clientY - rect.top) / rect.height) * 2 - 1);
};

const onPointerDown = (ev: PointerEvent) => {
  if (!editMode) return;
  getPointer(ev);
  raycaster.setFromCamera(pointer, camera);
  const hits = raycaster.intersectObjects(scene.children, true);
  const hit = hits.find((h) => (h.object as any).isMesh && (h.object as THREE.Mesh).userData?.placedIndex != null);
  if (!hit) return;
  selected = hit.object as THREE.Mesh;
  dragging = true;

  // horizontal plane at current Y
  dragPlane.constant = -selected.position.y;
  const pt = new THREE.Vector3();
  raycaster.ray.intersectPlane(dragPlane, pt);
  dragOffset.copy(pt).sub(selected.position);

  renderer.domElement.setPointerCapture(ev.pointerId);
};

const onPointerMove = (ev: PointerEvent) => {
  if (!editMode || !dragging || !selected) return;
  getPointer(ev);
  raycaster.setFromCamera(pointer, camera);
  const pt = new THREE.Vector3();
  if (!raycaster.ray.intersectPlane(dragPlane, pt)) return;

  const target = pt.sub(dragOffset);

  const { placedIndex, l, w } = selected.userData as { placedIndex: number; l: number; w: number };

  let newX = target.x;
  let newZ = target.z;

  const clampedX = Math.min(Math.max(newX, 0), cL - l);
  const clampedZ = Math.min(Math.max(newZ, 0), cW - w);

  const clamped = clampedX !== newX || clampedZ !== newZ;

  selected.position.x = clampedX;
  selected.position.z = clampedZ;

  // visual invalid cue
  const mat = selected.material as THREE.MeshStandardMaterial;
  if (clamped) {
    mat.emissive = new THREE.Color(0xef4444);
    mat.emissiveIntensity = 0.6;
    setEditWarning("Out of container bounds (clamped).");
  } else {
    mat.emissive = new THREE.Color(0x000000);
    mat.emissiveIntensity = 0.0;
    setEditWarning(null);
  }

  // live update placed array for UI (optional)
  if (onUpdatePlaced) {
    const next = placed.map((p, idx) => {
      if (idx !== placedIndex) return p;
      return { ...p, x: clampedX, z: clampedZ };
    });
    onUpdatePlaced(next);
  }
};

const onPointerUp = (ev: PointerEvent) => {
  if (!editMode) return;
  dragging = false;
  selected = null;
  try {
    renderer.domElement.releasePointerCapture(ev.pointerId);
  } catch {}
};

renderer.domElement.addEventListener("pointerdown", onPointerDown);
